function updateAI(ai,dt){ if(center.length < 10) return; /* ===================================================== 1. STABILER STRECKENPUNKT ===================================================== Die KI sucht nicht mehr die komplette Strecke ab. Dadurch springt sie bei Kreuzungen oder engen Stellen nicht plötzlich zu einem falschen Streckenpunkt. */ let best = null; const searchBack = 8; const searchForward = 38; for(let o=-searchBack;o<=searchForward;o++){ const index = ((ai.idx+o)%center.length+ center.length)% center.length; const p = pointAt(index); const d = Math.hypot( ai.x-p.x, ai.y-p.y ); /* Punkte hinter dem Auto bekommen eine kleine Strafe. So bleibt die KI eher auf ihrem aktuellen Fortschritt. */ const progressPenalty = o < 0 ? 18 : 0; const score = d+progressPenalty; if( !best || score < best.score ){ best = { i:index, d, score, x:p.x, y:p.y }; } } const near = best; ai.idx = near.i; const curve = calculateCurve(ai.idx); const car = cars[ai.carIndex]; /* ===================================================== 2. ZIELGESCHWINDIGKEIT ===================================================== */ let speedFactor = 0.94 - curve*0.48; speedFactor += (difficulty-2)*0.045; let desiredSpeed = car.top * speedFactor * ai.skill; /* Wenn die KI etwas neben der Strecke ist, wird sie vorsichtiger. */ if(near.d > roadWidth*.55){ desiredSpeed *= clamp( 1- (near.d-roadWidth*.55)/ roadWidth, .35, 1 ); } ai.targetSpeed = desiredSpeed; /* ===================================================== 3. FLÜSSIGE BESCHLEUNIGUNG ===================================================== Statt sofort stark zu beschleunigen oder zu bremsen, nähert sich die Geschwindigkeit langsam dem Ziel. */ const speedResponse = ai.speed < ai.targetSpeed ? 3.2 : 5.0; ai.speed += ( ai.targetSpeed- ai.speed ) * Math.min( 1, speedResponse*dt ); ai.speed = clamp( ai.speed, 0, car.top*1.02 ); /* ===================================================== 4. LOOKAHEAD ===================================================== */ const look = Math.floor( 7+ ai.speed*.075+ curve*5 ); const targetIndex = ai.idx+look; const target = pointAt(targetIndex); const tangent = tangentAt(targetIndex); /* ===================================================== 5. FAHRSPUR ===================================================== */ const normalX = -Math.sin(tangent); const normalY = Math.cos(tangent); /* In engen Kurven etwas weniger seitlicher Abstand. */ const safeLane = ai.laneOffset * roadWidth * .55 * (1-curve*.55); const tx = target.x+ normalX*safeLane; const ty = target.y+ normalY*safeLane; /* ===================================================== 6. ZIELRICHTUNG ===================================================== */ const targetDirection = Math.atan2( ty-ai.y, tx-ai.x ); /* Wir kombinieren Zielrichtung und Streckenrichtung. Dadurch fährt die KI nicht so nervös. */ const roadDirection = tangentAt( ai.idx+Math.floor(look*.35) ); let targetAngle = targetDirection*.72+ roadDirection*.28; /* Winkel normalisieren */ while( targetAngle-ai.angle > Math.PI ) targetAngle-=Math.PI*2; while( targetAngle-ai.angle < -Math.PI ) targetAngle+=Math.PI*2; /* ===================================================== 7. FLÜSSIGE LENKUNG ===================================================== Das ist die wichtigste Änderung gegen das Ruckeln. */ const angleError = angleDiff( ai.angle, targetAngle ); let targetSteer = clamp( angleError*1.35, -0.85, 0.85 ); /* Wenn wir zu weit draußen sind, stärker zur Strecke lenken. */ if( near.d > roadWidth*.55 ){ const correctionAngle = Math.atan2( near.y-ai.y, near.x-ai.x ); const correction = angleDiff( ai.angle, correctionAngle ); targetSteer = clamp( targetSteer+ correction*1.4, -1, 1 ); } /* Smooth Steering: Die Lenkung springt nicht mehr direkt auf den neuen Wert, sondern bewegt sich langsam dorthin. */ if( typeof ai.steer !== "number" ){ ai.steer = 0; } ai.steer += ( targetSteer- ai.steer ) * Math.min( 1, dt*5 ); /* ===================================================== 8. RICHTUNG KORRIGIEREN ===================================================== Kein plötzliches: ai.angle = roadAngle mehr. */ const roadAngle = tangentAt(ai.idx); const directionError = Math.abs( angleDiff( ai.angle, roadAngle ) ); if( directionError > 1.8 ){ const recoveryAngle = angleDiff( ai.angle, roadAngle ); ai.angle += recoveryAngle* Math.min( 1, dt*2.5 ); ai.speed += ( car.top*.28- ai.speed ) * Math.min( 1, dt*3 ); } /* ===================================================== 9. LENKUNG AUF DAS AUTO ANWENDEN ===================================================== */ const steeringSpeed = 1.55+ ai.speed/170; ai.angle += ai.steer* steeringSpeed* dt; /* ===================================================== 10. BEWEGUNG ===================================================== */ ai.x += Math.cos(ai.angle)* ai.speed* .30* dt; ai.y += Math.sin(ai.angle)* ai.speed* .30* dt; /* ===================================================== 11. STRECKENKORREKTUR ===================================================== Die KI wird sanft zurück zur Strecke gezogen. KEIN TELEPORTIEREN. */ const after = nearest( ai.x, ai.y ); if( after.d > roadWidth*.8 ){ const pullStrength = clamp( ( after.d- roadWidth*.45 )/ roadWidth, 0, 1 ); /* Geschwindigkeit reduzieren. */ ai.speed *= Math.pow( .88, dt*60 ); /* Sanft Richtung Strecke ziehen. */ ai.x += ( after.x- ai.x )* pullStrength* Math.min( 1, dt*3.5 ); ai.y += ( after.y- ai.y )* pullStrength* Math.min( 1, dt*3.5 ); } /* ===================================================== 12. BILDSCHIRM-SICHERHEIT ===================================================== Falls die KI durch einen Fehler außerhalb des sichtbaren Bereichs landet, wird sie nicht gelöscht. Sie wird nur wieder in Richtung Strecke gelenkt. */ const margin = 120; if( ai.x < -margin || ai.x > W+margin || ai.y < -margin || ai.y > H+margin ){ const safe = nearest( W/2, H/2 ); ai.x += ( safe.x- ai.x )* Math.min( 1, dt*1.5 ); ai.y += ( safe.y- ai.y )* Math.min( 1, dt*1.5 ); ai.speed *= .96; } /* ===================================================== 13. STUCK DETECTION ===================================================== Kein sichtbares Zurücksetzen mehr. Stattdessen wird die KI langsam wieder in Bewegung gebracht. */ const moved = Math.hypot( ai.x-ai.lastX, ai.y-ai.lastY ); if( moved < 1.0 && ai.speed > 8 ){ ai.stuck += dt; }else{ ai.stuck = Math.max( 0, ai.stuck-dt*.5 ); } ai.lastX = ai.x; ai.lastY = ai.y; /* Wenn die KI länger feststeckt: nur Geschwindigkeit und Richtung sanft korrigieren. KEIN Teleport. */ if( ai.stuck > 1.5 ){ const recovery = nearest( ai.x, ai.y ); const recoveryAngle = Math.atan2( recovery.y-ai.y, recovery.x-ai.x ); const recoveryDiff = angleDiff( ai.angle, recoveryAngle ); ai.angle += recoveryDiff* Math.min( 1, dt*2 ); ai.speed = Math.max( ai.speed, car.top*.18 ); ai.stuck = 0; } /* ===================================================== 14. KLEINE FEHLER ===================================================== */ if( difficulty===1 && Math.random()< .00003*dt*60 ){ ai.speed *= .97; } }
